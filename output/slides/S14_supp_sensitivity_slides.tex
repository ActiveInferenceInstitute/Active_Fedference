\subsection{Supplement: parameter-sensitivity
methods}\label{sec:supp-sensitivity}

\begin{frame}[allowframebreaks]{Supplement: parameter-sensitivity
methods}
This supplement documents how the sensitivity grid of
sec.~13.13 is generated and --- importantly for
a reader trying to reconcile numbers across studies --- where its two
sweeps use \emph{different} seeding and trial budgets. It answers two
questions the main section leaves implicit: exactly which seed drives
each cell (so any cell is independently reproducible), and why the
cross-study summary's sensitivity row is not directly comparable, at
matched trial counts, to the standalone heatmap.
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Experimental protocol for grid
sensitivity}
\protect\phantomsection\label{experimental-protocol-for-grid-sensitivity}
The sensitivity sweep is implemented in
\breaktt{fedference.experiments.run\_belief\_sharing\_sensitivity} and
\breaktt{fedference.experiments.run\_hierarchical\_sensitivity}. Each
function accepts a tuple of acuity values and a tuple of colony sizes.
In the \textbf{belief-sharing} sweep every (acuity, colony-size) cell
averages \(n_{\text{trials}}\) independent trials, each seeded via a
deterministic formula:

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Experimental protocol for grid
sensitivity}
\begin{equation}\protect\phantomsection\label{eq:sensitivity-seed}{
\text{seed}_{\text{cell}} = \text{seed}_{\text{base}} + i \cdot 10^5 + j \cdot 10^3 + t
}\end{equation}

The deterministic seed rule eq.~\ref{eq:sensitivity-seed} makes every
grid cell and trial independently reproducible from the base seed.

where \(i\) indexes acuity, \(j\) indexes colony size, and \(t\) indexes
the trial within a cell. For the belief-sharing sweep this guarantees
that:

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Experimental protocol for grid
sensitivity}
\begin{enumerate}
\tightlist
\item
  no two cells share a trial seed (no correlation between cells);
\item
  re-running with the same \texttt{seed\_base} is bit-identical
  (reproducibility);
\item
  different \texttt{seed\_base} values produce independent replicates
  (robustness checking).
\end{enumerate}

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Experimental protocol for grid
sensitivity}
The \textbf{hierarchical} sweep uses a simpler protocol: every cell
calls \breaktt{run\_hierarchical\_world} once with the same base seed
(its internal trials are seeded by that run), so hierarchical cells
share the base seed rather than the per-cell formula above.
\end{frame}

The 5×5 = 25 cells per system are run with \texttt{seed\_base\ =\ 0} by

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Belief-sharing condition in the
sensitivity grid}
\protect\phantomsection\label{belief-sharing-condition-in-the-sensitivity-grid}
Each trial in the belief-sharing sweep:

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Belief-sharing condition in the
sensitivity grid}
\begin{enumerate}
\tightlist
\item
  Draws a random true state and one noisy observation per agent (same
  protocol as \breaktt{run\_belief\_sharing}).
\item
  Runs one belief-sharing round with \breaktt{communicate=True}
  (communicating) and \breaktt{communicate=False} (isolated).
\item
  Records \texttt{mean\_accuracy} for each condition.
\item
  The cell value is the average over \texttt{n\_trials} of this gap.
\end{enumerate}
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Hierarchical POMDP condition in
the sensitivity grid}
\protect\phantomsection\label{hierarchical-pomdp-condition-in-the-sensitivity-grid}
Each cell in the hierarchical sweep calls
\breaktt{run\_hierarchical\_world} once with the cell's acuity and colony
size, passing the constant base seed (not the per-cell formula, which
applies only to the belief-sharing sweep). The returned
\breaktt{location\_accuracy\_gap} (hierarchical minus flat) becomes the
cell value.
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Figure rendering for sensitivity
summaries}
\protect\phantomsection\label{figure-rendering-for-sensitivity-summaries}
\breaktt{generate\_sensitivity\_heatmap} assembles the two grids into a
1×2 matplotlib \texttt{imshow} figure with RdYlGn colormap, symmetric
bounds at \(\pm\max(|\text{gap}|)\), per-cell numeric annotations, and a
per-panel colorbar labeled ``Accuracy gap (hierarchical/comm. −
baseline)''. The figure is written to
\breaktt{output/figures/sensitivity\_heatmap.png}.
\end{frame}

The robustness row uses 40 matched trials per seed and rate; the

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Cross-study summary
construction}
The Study 8 row below uses 3 trials per cell --- smaller than the
full-resolution 20-trial \breaktt{Trials\ per\ cell} grid documented
above for the standalone sensitivity heatmap figure, a deliberate
runtime budget for the per-seed cross-study loop rather than an
oversight --- so the two are not directly comparable at matched trial
counts.

\end{frame}
