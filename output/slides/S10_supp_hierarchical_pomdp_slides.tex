\subsection{Supplement: hierarchical POMDP methods and
parameters}\label{sec:supp-hierarchical}

\begin{frame}[allowframebreaks]{Supplement: hierarchical POMDP methods
and parameters}
This supplement makes the two-level construction of
sec.~13.9 concrete: how location (L1) is
coupled to context (L2) through context-conditioned priors, what the
alternating-minimization update actually computes, and the exact
parameters of the executed run. It answers the question the main section
brackets --- \emph{by what mechanism does a second latent level enter
the inference at all} --- and thereby fixes why the hierarchy resolves
context above chance while leaving location accuracy statistically
unchanged.
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Generative model for
context-gated location inference}
\protect\phantomsection\label{generative-model-for-context-gated-location-inference}
The two-level POMDP implemented in
\breaktt{fedference.pomdp.build\_hierarchical\_world} couples the
sentinel's 9-location L1 factor to a 2-state L2 context factor via
\textbf{context-conditioned L1 priors}:

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Generative model for
context-gated location inference}
\begin{itemize}
\tightlist
\item
  \textbf{L1 (location)} --- the standard 3x3 grid of
  \breaktt{build\_sentinel\_world} with \texttt{n\_s\ =} 9 states and
  sensor acuity 0.85;
\item
  \textbf{L2 (context)} --- a binary state (\texttt{quiet} /
  \texttt{alert}) with a symmetric transition matrix (persistence 0.90)
  and an initial uniform prior;
\item
  \textbf{L1 priors given context} --- \texttt{quiet}: uniform over all
  9 cells; \texttt{alert}: mass 0.60 at the center cell (the den), the
  residual spread uniformly.
\end{itemize}
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Inference algorithm for top-down
empirical priors}
\protect\phantomsection\label{inference-algorithm-for-top-down-empirical-priors}
\breaktt{fedference.pomdp.hierarchical\_infer} performs 4 passes of
alternating minimization:

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Inference algorithm for top-down
empirical priors}
\begin{enumerate}
\tightlist
\item
  \textbf{L2 → L1 empirical prior}: \(\widetilde{\pi}_{0,\mathrm{L1}} =
  \sum_c q_{\text{ctx}}[c]\,\pi_{0,\mathrm{L1}\mid c}\) (a soft mixture
  of the two context-conditioned priors).
\item
  \textbf{L1 update}: one-step variational posterior
  \(q_{\text{loc}} = \operatorname{softmax}(\log \widetilde{\pi}_{0,\mathrm{L1}} +
  \log A[\text{obs},\,\cdot])\).
\item
  \textbf{L1 → L2 marginal evidence}:
  \(\ell_c = \log\bigl(\pi_{0,\mathrm{L1}\mid c}^{\top}
  A[\text{obs},\,\cdot]\bigr)\) (evidence for context \(c\) from the
  observed location likelihood).
\item
  \textbf{L2 update}:
  \(q_{\text{ctx}} = \operatorname{softmax}(\log \pi_{0,\mathrm{L2}} + \ell)\).
\end{enumerate}

\end{frame}

\begin{frame}[fragile,allowframebreaks]{Inference algorithm for top-down
empirical priors}
After 4 iterations the agent broadcasts both \(q_{\text{loc}}\) and
\(q_{\text{ctx}}\); the colony federates each level independently via a
log-linear pool (eq.~6).
\end{frame}
