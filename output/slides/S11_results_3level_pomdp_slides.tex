\begin{frame}{Three-level hierarchical POMDP: an executed test of the
N-level template}
\protect\phantomsection\label{sec:results-3level}
The 2-level hierarchical POMDP (sec.~14.9)
couples location inference to a single global context.
\end{frame}

\begin{frame}[fragile]{Three-level hierarchical\ldots{} (part 2)}
The N-level architecture uses \texttt{build\_nlevel\_world} from the
\texttt{fedference.pomdp} module and provides a parameterized stack of
levels; the canonical 3-level example couples location (L1; 9 states) to
a context variable (L2;
\end{frame}

\begin{frame}[fragile]{Three-level hierarchical\ldots{} (part 3)}
2 states: \texttt{quiet} / \texttt{alert}) and further to a meta-context
variable (L3; 2 states: \texttt{low\_threat} / \texttt{high\_threat})
that gates the L2 prior.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 4)}
\begin{equation}\tag{38}\protect\phantomsection\label{eq:l3-to-l2-message}{
\tilde{D}_{\text{L2}} = \sum_k q_{\text{L3}}[k]\,p_{\text{L2|L3}}[k]
}\end{equation}
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 5)}
\begin{equation}\tag{39}\protect\phantomsection\label{eq:l2-to-l1-message}{
\tilde{D}_{\text{L1}} = \sum_c q_{\text{L2}}[c]\,p_{\text{L1|L2}}[c]
}\end{equation}
\end{frame}

\begin{frame}[fragile]{Three-level hierarchical\ldots{} (part 6)}
The inference algorithm (\texttt{fedference.pomdp.nlevel\_infer})
performs 4 passes of top-down / bottom-up alternating minimization: the
top-down pass propagates empirical priors from L3 \texorpdfstring{\ensuremath{\to}}{->} L2 \texorpdfstring{\ensuremath{\to}}{->} L1 via
eq.~38 and eq.~39;
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 7)}
the bottom-up pass updates each level's belief from the marginal
evidence contributed by the level below.

We compare two conditions over 960 seeded trials with 4 agents at sensor
acuity 0.85.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 8)}
\textbf{Flat condition.} Agents ignore all hierarchy and infer location
under a uniform prior.

\textbf{Three-level condition.} Agents run 4 alternating-minimization
iterations across all three levels before federating.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 9)}
The measured location accuracies are 0.984 (flat) and 0.966 (3-level), a
gap of -0.019.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 10)}
Across 128 independent seeds the 3-level location accuracy is 0.976 (SD
0.005; 95 \% CI 0.976--0.977) versus flat 0.981 (95 \% CI 0.980--0.981),
a mean accuracy gap of -0.004 (95 \% CI -0.005---0.003; Wilcoxon
signed-rank \(p = 2.88 \times 10^{-12}\), effect size \(r = 0.724\),
medium).
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 11)}
The signed mean gap and its interval describe the location-accuracy cost
of this configured three-level condition relative to the flat baseline.

The 3-level condition additionally reports context accuracy 0.697 and
meta-context accuracy 0.547.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 12)}
Against the two-state chance baseline of \(0.5\), location is recovered
and the intermediate context latent is resolved well above chance, but
the meta-context latent is only marginally above chance --- the weakest
of the three levels --- and is therefore \emph{not} convincingly
recovered here.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 13)}
The study thus demonstrates that the generic \(N\)-level
alternating-minimization runs and federates end-to-end and recovers the
fastest (location) and intermediate (context) latents; full recovery of
the slowest (meta-context) level is left open.
\end{frame}

\begin{frame}{Three-level hierarchical\ldots{} (part 14)}
The full figure comparing 2-level and 3-level belief dynamics is

on these results, see sec.~14.13.
\end{frame}
